// Retrieves all pending events at once and returns TRUE when at least one was
// pending. Replaces the events of the previous Poll, query them with Received.

METHOD PUBLIC Poll: BOOL
VAR
	pending:	DWORD;
END_VAR

////////////////////////////////

events := 0;

IF handle = 0 THEN
	Poll := FALSE;
	RETURN;
END_IF

IF Check(iox2plc_listener_poll_cext(
	port := handle,
	out_events := ADR(pending)
)) THEN
	events := pending;
END_IF

Poll := events <> 0;
