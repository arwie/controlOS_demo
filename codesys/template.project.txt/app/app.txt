PROGRAM app

VAR_OUTPUT
	cfg: AppCfg;
	cmd: AppCmd;
END_VAR
VAR_INPUT
	fbk: AppFbk;
END_VAR

VAR CONSTANT
	CFG_FILE: STRING := '/run/codesys/cfg';
END_VAR
VAR
	cmd_subscriber: IOX2.Subscriber('codesys/cmd', 'AppCmd', SIZEOF(AppCmd), Alignment:=8, Publishers:=1, Subscribers:=1, History:=FALSE);
	fbk_publisher:  IOX2.Publisher ('codesys/fbk', 'AppFbk', SIZEOF(AppFbk), Alignment:=8, Publishers:=1, Subscribers:=8, History:=FALSE);
	fbk_notifier:   IOX2.Notifier  ('codesys/fbk');
END_VAR

////////////////////////////////

cmd_subscriber.Receive(cmd);


IF init.done THEN
	main();
ELSE
	init();
END_IF


fbk_publisher.Send(fbk);
fbk_notifier.Notify();
